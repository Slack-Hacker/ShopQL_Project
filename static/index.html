🔍 1. Search & Filter

            SHOW PRODUCTS WHERE RATING >= 4.5 ORDER BY PRICE ASC;
            FIND PRODUCTS WHERE NAME CONTAINS "Pro";
            FIND PRODUCTS WHERE CATEGORY LIKE "ELEC%";
            FIND PRODUCTS WHERE PRICE BETWEEN 50 AND 500;
          


          

            
📦 2. Inventory Transactions

            ADD PRODUCT "Laptop" PRICE 55000;
            BUY PRODUCT "Laptop" QUANTITY 2;
            RESTOCK PRODUCT "Laptop" QUANTITY 10;
            UPDATE PRODUCT "Laptop" SET PRICE = 49999;
            DELETE PRODUCT "Laptop";
          


          

            
⚙️ 3. Logical Operators

            WHERE (CATEGORY = "ELECTRONICS" OR CATEGORY = "BOOKS") AND PRICE < 500;
